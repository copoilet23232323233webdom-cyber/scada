        public static void InitializeAlarms(Model model)
        {
            Alerts.Initialize(model);

            //CreateEmailSharedGroups();

            foreach (var device in model.Settings.ModbusClients)
            {
                switch (device.DeviceType)
                {
                    case Globals.DeviceType.Ppc:
                        InitializePPCAlarms(device, model.Settings);
                        break;

                    case Globals.DeviceType.Inverter:
                        if(device.Id.Contains("Master"))
                        {
                            InitializeInverterAlarms(device, model.Settings, PowerElectronicsInverter.DeviceModel.MASTER);
                        }
                        else
                        {
                            InitializeInverterAlarms(device, model.Settings, PowerElectronicsInverter.DeviceModel.SLAVE);
                        }
                        break;

                    case Globals.DeviceType.MeteoSensor:
                        InitializeMeteoAlarms(device, model.Settings);
                        break;

                    case Globals.DeviceType.IOModule:
                        if(device.GroupName == "ANM")
                        {
                            InitializeIOLogikAlarms(device, model.Settings);
                        }
                        break;

                    case Globals.DeviceType.Meter:
                        InitializeMainMeterAlarms(device, model.Settings);
                        break;

                    case Globals.DeviceType.CombinerBox:
                        InitializeCombinerBoxAlarms(device, model.Settings);
                        break;
                }
            }
        }

        /////////////// PPC
        public static void InitializePPCAlarms(ModbusMap map, Settings settings)
        {
            // Set alarms
            ModbusRegister irradiance = RadiationSensor.FindFirstPoaOnline(settings);

            ModbusRegister reg = map.ModbusRegisters.Find(x => x.Meta == "POWER_AC");

            int timeout = 10;

            reg.Alarms = new List<WebdomEvent>()
                {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                if(irradiance.Value == null) return false;

                                return ((Single)code)<=0 && (Double)irradiance.ScaledValue > 100;
                            },
                            EventTitle =$"PPC generation is 0 and solar radiation > 100W/m2, timeout={timeout} minutes",
                            Message = "PPC generation is 0 and solar radiation > 100W/m2",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                            RegisterKey = reg.Key,
                            BelongsTo = map.Id,
                            Plant = settings.Parameters.PlantName,
                            EventType =  WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.GENERATION,
                            EventPriority = EventPriority.PPC_LEVEL,
                            ShareGroups = new(){ShareGroup.WEBDOM }
                        }
            };

            reg = map.ModbusRegisters.Find(x => x.Meta == "DEVICE_STATUS");
            timeout = 15;
            reg.Alarms = new List<WebdomEvent>()
                {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                return (Convert.ToInt16(code)) == (ushort)Globals.DeviceStatus.ERROR ? true : false;
                            },
                            EventTitle = $"{map.Id} does not respond, communication failure, timeout={timeout} minutes",
                            Message = $"{map.Id} no responde, fallo de comunicación",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                            RegisterKey = reg.Key,
                            BelongsTo = map.Id,
                            Plant = settings.Parameters.PlantName,
                            EventType = WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.COMMUNICATION,
                            EventPriority = EventPriority.PPC_LEVEL,
                            ShareGroups = new(){ ShareGroup.WEBDOM }
                        }
            };

        }


        ////////////// INVERTERS
        public static void InitializeInverterAlarms(ModbusMap map, Settings settings, PowerElectronicsInverter.DeviceModel model)
        {

            ModbusRegister deviceStatus = map.ModbusRegisters.Find(x => x.Meta == "DEVICE_STATUS");
            int timeout = 15;
            switch (model)

                case PowerElectronicsInverter.DeviceModel.MASTER:
                    deviceStatus.Alarms = new List<WebdomEvent>()
                    {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                return (Convert.ToInt16(code)) == (ushort)Globals.DeviceStatus.ERROR ? true : false;
                            },
                            EventTitle = $"{map.Id} does not respond, communication failure, timeout={timeout} minutes",
                            Message = $"{map.Id} does not respond, communication failure",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                            RegisterKey = deviceStatus.Key,
                            BelongsTo = map.Id,
                            Plant = settings.Parameters.PlantName,
                            EventType = WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.COMMUNICATION,
                            EventPriority = EventPriority.POWERSTATION_LEVEL,
                            ShareGroups = new(){ ShareGroup.WEBDOM }
                        }
                    };

                    break;

                case PowerElectronicsInverter.DeviceModel.SLAVE:
                    deviceStatus.Alarms = new List<WebdomEvent>()
                    {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                return (Convert.ToInt16(code)) == (ushort)Globals.DeviceStatus.ERROR ? true : false;
                            },
                            EventTitle = $"{map.Id} does not respond, communication failure, timeout={timeout} minutes",
                            Message = $"{map.Id} does not respond, communication failure",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                            RegisterKey = deviceStatus.Key,
                            BelongsTo = map.Id,
                            Plant = settings.Parameters.PlantName,
                            EventType = WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.COMMUNICATION,
                            EventPriority = EventPriority.POWERSTATION_LEVEL,
                            ShareGroups = new(){ ShareGroup.WEBDOM }
                        }
                    };

                    break;

            }


            ModbusRegister reg = map.ModbusRegisters.Find(x => x.Meta == "POWER_AC");
            timeout = 5;
            switch (model)
            {
                case PowerElectronicsInverter.DeviceModel.MASTER:
                    reg.Alarms = new List<WebdomEvent>()
                    {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                if(irradiance.Value == null) return false;

                                return ((Int16)code)<=0 && (Double)irradiance.ScaledValue > 100;
                            },
                            EventTitle = $"{map.Id} generation is 0 and solar radiation > 100W/m2, timeout={timeout} minutes",
                            Message = $"{map.Id} generation is 0 and solar radiation > 100W/m2",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                            RegisterKey = reg.Key,
                            BelongsTo = map.Id,
                            Plant = settings.Parameters.PlantName,
                            EventType = WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.GENERATION,
                            EventPriority = EventPriority.POWERSTATION_LEVEL,
                            ShareGroups = new(){ ShareGroup.WEBDOM},
                            DeviceStatus = deviceStatus
                        }
                    };

                    break;

                case PowerElectronicsInverter.DeviceModel.SLAVE:
                    reg.Alarms = new List<WebdomEvent>()
                    {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                if(irradiance.Value == null) return false;

                                return ((Int16)code)<=0 && (Double)irradiance.ScaledValue > 350;
                            },
                            EventTitle = $"{map.Id} generation is 0 and solar radiation > 350W/m2, timeout={timeout} minutes",
                            Message = $"{map.Id} generation is 0 and solar radiation > 350W/m2",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                            RegisterKey = reg.Key,
                            BelongsTo = map.Id,
                            Plant = settings.Parameters.PlantName,
                            EventType = WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.GENERATION,
                            EventPriority = EventPriority.POWERSTATION_LEVEL,
                            ShareGroups = new(){ ShareGroup.WEBDOM},
                            DeviceStatus = deviceStatus
                        }
                    };

                    break;
            }

            /// Modules (not Master) will generate exception cause no CURRENT_ERROR channel
            if (model == PowerElectronicsInverter.DeviceModel.MASTER)
            {
                ModbusRegister reg1 = map.ModbusRegisters.Find(x => x.Meta == "CURRENT_ERROR");
                timeout = 5;
                WebdomEvent e1 = new()
                {
                    EventTitle = $"Inverter with failure code, timeout={timeout} minutes",
                    TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                    RegisterKey = reg1.Key,
                    BelongsTo = map.Id,
                    Plant = settings.Parameters.PlantName,
                    EventType = WebdomControl.EventType.Alert,
                    Where = map.GroupName,
                    Title = "Automated alert",
                    DeviceType = map.DeviceType,
                    EventGroup = EventGroup.DEVICE_FAULT,
                    EventPriority = EventPriority.POWERSTATION_LEVEL,
                    ShareGroups = new() { ShareGroup.WEBDOM},
                    DeviceStatus = deviceStatus
                };

                e1.TriggerCondition = delegate (Object code)
                {
                    e1.Message = $"{map.Id} has failure code {reg1.Value}";
                    return (ushort)code != 0;
                };

                reg1.Alarms = new List<WebdomEvent>() { e1 };



                ModbusRegister reg2 = map.ModbusRegisters.Find(x => x.Meta == "IGBT_TEMP");
                timeout = 10;
                WebdomEvent e2 = new()
                {
                    EventTitle = "IGBT high temperature alarm, timeout={timeout} minutes",
                    TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                    RegisterKey = reg2.Key,
                    BelongsTo = map.Id,
                    Plant = settings.Parameters.PlantName,
                    EventType = WebdomControl.EventType.Alert,
                    Where = map.GroupName,
                    Title = "Automated alert",
                    DeviceType = map.DeviceType,
                    EventGroup = EventGroup.THRESHOLD,
                    EventPriority = EventPriority.POWERSTATION_LEVEL,
                    ShareGroups = new() { ShareGroup.WEBDOM},
                    DeviceStatus = deviceStatus
                };

                e2.TriggerCondition = delegate (Object code)
                {
                    e2.Message = $"{map.Id} with channel {reg2.Meta} with temperature {reg2.Value}";
                    return (short)code > 116 && (short)code < 121;
                };

                timeout = 1;
                WebdomEvent e3 = new()
                {
                    EventTitle = $"IGBT derated because of too high temperature, timeout={timeout} minutes",
                    TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                    RegisterKey = reg2.Key,
                    BelongsTo = map.Id,
                    Plant = settings.Parameters.PlantName,
                    EventType = WebdomControl.EventType.Alert,
                    Where = map.GroupName,
                    Title = "Automated alert",
                    DeviceType = map.DeviceType,
                    EventGroup = EventGroup.THRESHOLD,
                    EventPriority = EventPriority.POWERSTATION_LEVEL,
                    ShareGroups = new() { ShareGroup.WEBDOM},
                    DeviceStatus = deviceStatus
                };

                e3.TriggerCondition = delegate (Object code)
                {
                    e3.Message = $"{map.Id} with channel {reg2.Meta} with temperature {reg2.Value}-->Derating";
                    return (short)code >= 121;
                };

                reg2.Alarms = new List<WebdomEvent>() { e2, e3 };

            }

        }

        public static void InitializeIOLogikAlarms(ModbusMap map, Settings settings)
        {
            ModbusRegister deviceStatus = map.ModbusRegisters.Find(x => x.Meta == "DEVICE_STATUS");

            int timeout = 15;
            deviceStatus.Alarms = new List<WebdomEvent>()
                {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                return (Convert.ToInt16(code)) == (ushort)Globals.DeviceStatus.ERROR ? true : false;
                            },
                            EventTitle = $"{map.Id} does not respond, communication failure, Timeout: {timeout} minutes",
                            Message = $"{map.Id} does not respond, communication failure",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),

                            Plant = settings.Parameters.PlantName,
                            EventType = WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.COMMUNICATION,
                            EventPriority = EventPriority.POWERSTATION_LEVEL,
                            ShareGroups = new(){ ShareGroup.WEBDOM }
                        }
            };


            // Alert when channel 0 (first setpoint input) changes
            ModbusRegister reg = map.ModbusRegisters.Find(x => x.Meta == "AI_0");

            double triggerDifferenceInMW = 0.1;


            WebdomEvent e1 = new WebdomEvent()
            {
                EventTitle = $"{map.Id} - New ANM setpoint (Input 1), Timeout: Immediate",
                Message = $"{map.Id} - New ANM setpoint (Input 1) --> {reg.Value} MW",
                TimeoutBeforeTrigger = null, // no timeout, immediate trigger
                RegisterKey = reg.Key,
                BelongsTo = map.Id,
                Plant = settings.Parameters.PlantName,
                EventType = WebdomControl.EventType.Alert,
                Where = map.GroupName,
                Title = "Automated alert",
                DeviceType = map.DeviceType,
                EventGroup = EventGroup.COMMUNICATION,
                EventPriority = EventPriority.POWERSTATION_LEVEL,
                ShareGroups = new() { ShareGroup.WEBDOM },
                DeviceStatus = deviceStatus
            };

            {

                    {
                        e1.Message = $"{map.Id} - New ANM setpoint (Input 1) --> {Convert.ToSingle(reg.Value).ToString("#.##")} MW";

                    {
                        return false;
                    }
                }

                    return false;
                }
            };

                EventPriority = EventPriority.POWERSTATION_LEVEL,
                ShareGroups = new() { ShareGroup.WEBDOM },
                DeviceStatus = deviceStatus

            e2.TriggerConditionEx = delegate (Object code, Object previousValue, Object scaledValue)

        ////////////// COMBINER BOX ALARMS
        public static void InitializeCombinerBoxAlarms(ModbusMap map, Settings settings)
        {
            ModbusRegister irradiance = RadiationSensor.FindFirstPoaOnline(settings);

            ModbusRegister deviceStatus = map.ModbusRegisters.Find(x => x.Meta == "DEVICE_STATUS");
            int timeout = 25;
            deviceStatus.Alarms = new List<WebdomEvent>()
                {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                if(irradiance.Value == null) return false;

                                return ((Convert.ToInt16(code)) == (ushort)Globals.DeviceStatus.ERROR) && (double)irradiance.ScaledValue > 100 ? true : false;
                            },
                            EventTitle = $"{map.Id} does not respond, communication failure with radiation >100W/m2, timeout={timeout} minutes",
                            Message = $"{map.Id} does not respond, communication failure with radiation >100W/m2",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                            RegisterKey = deviceStatus.Key,
                            BelongsTo = map.Id,
                            Plant = settings.Parameters.PlantName,
                            EventType = WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.COMMUNICATION,
                            EventPriority = EventPriority.COMBINERBOX_LEVEL,
                            ShareGroups = new(){ ShareGroup.WEBDOM, ShareGroup.AREA, ShareGroup.AREA_RESPONSIBLE, ShareGroup.CENTROCONTROL1, ShareGroup.CENTROCONTROL2 }
                        }
            };

            ModbusRegister reg = map.ModbusRegisters.Find(x => x.Meta == "CURRENT_DC");

            reg.Alarms = new List<WebdomEvent>()
                {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                if(irradiance.Value == null) return false;

                                return (Single)code < 5 && (double)irradiance.ScaledValue > 300 ? true : false;
                            },
                            EventTitle = $"{map.Id} wit generation 0 , and radiation >300W/m2, timeout={timeout} minutes",
                            Message = $"{map.Id} wit generation 0 , and radiation >300W/m2",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                            RegisterKey = reg.Key,
                            BelongsTo = map.Id,
                            Plant = settings.Parameters.PlantName,
                            EventType = WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.GENERATION,
                            EventPriority = EventPriority.COMBINERBOX_LEVEL,
                            ShareGroups = new(){ ShareGroup.WEBDOM, ShareGroup.AREA, ShareGroup.AREA_RESPONSIBLE, ShareGroup.CENTROCONTROL1, ShareGroup.CENTROCONTROL2 },
                            DeviceStatus = deviceStatus
                        }
            };

            timeout = 30;
            foreach (var r in map.ModbusRegisters)
            {
                if (r.Meta.Contains("PV_CURRENT"))
                {
                    r.Alarms = new List<WebdomEvent>()
                    {
                        new WebdomEvent()
                        {
                            TriggerCondition = delegate (Object code)
                            {
                                if(irradiance.Value == null) return false;
                                                     //current expressed in mA 
                                return (ushort)code <= 1000 &&  (double)irradiance.ScaledValue > 300 ? true : false;
                            },
                            EventTitle = $"{map.Id} - String {r.Meta} generation 0, with solar radiation >300W/m2, timeout={timeout} minutes",
                            Message = $"{map.Id} - String {r.Meta} generation 0, with solar radiation >300W/m2",
                            TimeoutBeforeTrigger = TimeSpan.FromMinutes(timeout),
                            RegisterKey = r.Key,
                            BelongsTo = map.Id,
                            Plant = settings.Parameters.PlantName,
                            EventType = WebdomControl.EventType.Alert,
                            Where = map.GroupName,
                            Title = "Automated alert",
                            DeviceType = map.DeviceType,
                            EventGroup = EventGroup.GENERATION,
                            EventPriority = EventPriority.STRING_LEVEL,
                            ShareGroups = new(){ ShareGroup.WEBDOM, ShareGroup.AREA, ShareGroup.AREA_RESPONSIBLE, ShareGroup.CENTROCONTROL1, ShareGroup.CENTROCONTROL2 },
                            DeviceStatus = deviceStatus
                        }
                    };

                }
            }
